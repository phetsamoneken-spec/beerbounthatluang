1. ຄັດລອກໂຄດ Code.gs ດ້ານລຸ່ມນີ້:

const SPREADSHEET_ID = "1Vz1hNNIMl1bw6wF4gDlZymv0YFiqYw21q5VuJ7Siwss";

function doGet(e) {
  return HtmlService.createTemplateFromFile('index')
    .evaluate()
    .setTitle('ລະບົບລົງທະບຽນ ແລະ ກວດສອບ QR Code')
    .addMetaTag('viewport', 'width=device-width, initial-scale=1')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function getSheet() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName("Sheet1");
  if (!sheet) {
    sheet = ss.insertSheet("Sheet1");
    sheet.appendRow(["ວັນທີເວລາ", "ID", "ຊື່ ແລະ ນາມສະກຸນ", "ເບີໂທ", "ປະເພດຮ້ານ", "ລາຍລະອຽດຂອງແຖມ", "ຮູບບັດ (Base64)", "ຮູບຮ້ານ (Base64)", "ສະຖານະ"]);
  }
  return sheet;
}

function registerUser(data) {
  try {
    const sheet = getSheet();
    const timestamp = new Date();
    const rand = Math.floor(100000 + Math.random() * 900000);
    const regId = "REG-" + rand;
    sheet.appendRow([timestamp, regId, data.fullName, data.phone, data.branch, data.rewards, data.idCardImage, data.shopImage, "ຍັງບໍ່ທັນຮັບ"]);
    return { success: true, regId: regId };
  } catch (e) { return { success: false, error: e.toString() }; }
}

function getUserData(regId) {
  try {
    const sheet = getSheet();
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][1] === regId) {
        return { success: true, regId: rows[i][1], fullName: rows[i][2], phone: rows[i][3], branch: rows[i][4], rewards: rows[i][5], idCardImage: rows[i][6], shopImage: rows[i][7], status: rows[i][8] };
      }
    }
    return { success: false, error: "ບໍ່ພົບຂໍ້ມູນ" };
  } catch (e) { return { success: false, error: e.toString() }; }
}

function verifyReceipt(regId) {
  try {
    const sheet = getSheet();
    const rows = sheet.getDataRange().getValues();
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][1] === regId) {
        sheet.getRange(i + 1, 9).setValue("ຮັບເຄື່ອງແລ້ວ ✅");
        return { success: true };
      }
    }
    return { success: false, error: "ບໍ່ພົບຂໍ້ມູນ" };
  } catch (e) { return { success: false, error: e.toString() }; }
}
